"""
A simple tokenizer and recursive descent parser for arithmetic expressions.
Supports integers, parentheses, and the four basic operators (+, -, *, /).
"""

from enum import Enum
from dataclasses import dataclass
from typing import List, Optional


class TokenType(Enum):
    """Token types for the arithmetic expression language."""
    INTEGER = "INTEGER"
    PLUS = "PLUS"
    MINUS = "MINUS"
    MULTIPLY = "MULTIPLY"
    DIVIDE = "DIVIDE"
    LPAREN = "LPAREN"
    RPAREN = "RPAREN"
    EOF = "EOF"


@dataclass
class Token:
    """Represents a single token in the expression."""
    type: TokenType
    value: Optional[str] = None

    def __repr__(self):
        return f"Token({self.type.value}, {self.value})"


class Tokenizer:
    """
    Converts a string input into a sequence of tokens.
    Handles whitespace, integers, and operator symbols.
    """

    def __init__(self, text: str):
        """
        Initialize the tokenizer with input text.

        Args:
            text: The arithmetic expression string to tokenize.
        """
        self.text = text
        self.pos = 0
        self.current_char = self.text[0] if text else None

    def advance(self):
        """Move to the next character in the input."""
        self.pos += 1
        if self.pos >= len(self.text):
            self.current_char = None
        else:
            self.current_char = self.text[self.pos]

    def skip_whitespace(self):
        """Skip whitespace characters."""
        while self.current_char is not None and self.current_char.isspace():
            self.advance()

    def read_integer(self) -> str:
        """
        Read a multi-digit integer from the input.

        Returns:
            String representation of the integer.
        """
        result = ""
        while self.current_char is not None and self.current_char.isdigit():
            result += self.current_char
            self.advance()
        return result

    def tokenize(self) -> List[Token]:
        """
        Tokenize the entire input string.

        Returns:
            List of tokens including EOF token.
        """
        tokens = []

        while self.current_char is not None:
            # Skip whitespace
            if self.current_char.isspace():
                self.skip_whitespace()
                continue

            # Read integer
            if self.current_char.isdigit():
                integer = self.read_integer()
                tokens.append(Token(TokenType.INTEGER, integer))
                continue

            # Operators and parentheses
            char = self.current_char
            if char == "+":
                tokens.append(Token(TokenType.PLUS))
                self.advance()
            elif char == "-":
                tokens.append(Token(TokenType.MINUS))
                self.advance()
            elif char == "*":
                tokens.append(Token(TokenType.MULTIPLY))
                self.advance()
            elif char == "/":
                tokens.append(Token(TokenType.DIVIDE))
                self.advance()
            elif char == "(":
                tokens.append(Token(TokenType.LPAREN))
                self.advance()
            elif char == ")":
                tokens.append(Token(TokenType.RPAREN))
                self.advance()
            else:
                raise ValueError(f"Invalid character: {char}")

        tokens.append(Token(TokenType.EOF))
        return tokens


@dataclass
class ASTNode:
    """Base class for abstract syntax tree nodes."""
    pass


@dataclass
class Number(ASTNode):
    """Represents a numeric literal."""
    value: int


@dataclass
class BinaryOp(ASTNode):
    """Represents a binary operation (e.g., 2 + 3)."""
    left: ASTNode
    op: Token
    right: ASTNode


class Parser:
    """
    Recursive descent parser for arithmetic expressions.
    Implements operator precedence: multiplication and division bind
    tighter than addition and subtraction.
    """

    def __init__(self, tokens: List[Token]):
        """
        Initialize the parser with a token sequence.

        Args:
            tokens: List of tokens from the tokenizer.
        """
        self.tokens = tokens
        self.pos = 0
        self.current_token = tokens[0] if tokens else None

    def advance(self):
        """Move to the next token."""
        self.pos += 1
        if self.pos >= len(self.tokens):
            self.current_token = Token(TokenType.EOF)
        else:
            self.current_token = self.tokens[self.pos]

    def eat(self, token_type: TokenType):
        """
        Consume a token of the specified type.

        Args:
            token_type: Expected token type.

        Raises:
            AssertionError: If current token doesn't match expected type.
        """
        assert self.current_token.type == token_type, \
            f"Expected {token_type}, got {self.current_token.type}"
        self.advance()

    def factor(self) -> ASTNode:
        """
        Parse a factor: integer or parenthesized expression.

        Grammar: factor : INTEGER | LPAREN expr RPAREN
        """
        token = self.current_token

        if token.type == TokenType.INTEGER:
            self.eat(TokenType.INTEGER)
            return Number(int(token.value))
        elif token.type == TokenType.LPAREN:
            self.eat(TokenType.LPAREN)
            node = self.expr()
            self.eat(TokenType.RPAREN)
            return node

        raise ValueError(f"Invalid factor: {token}")

    def term(self) -> ASTNode:
        """
        Parse a term: handles multiplication and division.

        Grammar: term : factor ((MULTIPLY | DIVIDE) factor)*
        """
        node = self.factor()

        while self.current_token.type in (TokenType.MULTIPLY, TokenType.DIVIDE):
            token = self.current_token
            if token.type == TokenType.MULTIPLY:
                self.eat(TokenType.MULTIPLY)
            else:
                self.eat(TokenType.DIVIDE)

            node = BinaryOp(left=node, op=token, right=self.factor())

        return node

    def expr(self) -> ASTNode:
        """
        Parse an expression: handles addition and subtraction.

        Grammar: expr : term ((PLUS | MINUS) term)*
        """
        node = self.term()

        while self.current_token.type in (TokenType.PLUS, TokenType.MINUS):
            token = self.current_token
            if token.type == TokenType.PLUS:
                self.eat(TokenType.PLUS)
            else:
                self.eat(TokenType.MINUS)

            node = BinaryOp(left=node, op=token, right=self.term())

        return node

    def parse(self) -> ASTNode:
        """
        Parse the complete expression.

        Returns:
            Abstract syntax tree representing the expression.
        """
        node = self.expr()
        if self.current_token.type != TokenType.EOF:
            raise ValueError("Invalid expression: unexpected tokens after end")
        return node


class Evaluator:
    """
    Evaluates an abstract syntax tree and returns the result.
    """

    def evaluate(self, node: ASTNode) -> int:
        """
        Recursively evaluate an AST node.

        Args:
            node: AST node to evaluate.

        Returns:
            Integer result of evaluation.
        """
        if isinstance(node, Number):
            return node.value

        if isinstance(node, BinaryOp):
            left_val = self.evaluate(node.left)
            right_val = self.evaluate(node.right)

            if node.op.type == TokenType.PLUS:
                return left_val + right_val
            elif node.op.type == TokenType.MINUS:
                return left_val - right_val
            elif node.op.type == TokenType.MULTIPLY:
                return left_val * right_val
            elif node.op.type == TokenType.DIVIDE:
                if right_val == 0:
                    raise ValueError("Division by zero")
                return int(left_val / right_val)

        raise ValueError(f"Unknown node type: {type(node)}")


def evaluate_expression(text: str) -> int:
    """
    Tokenize, parse, and evaluate an arithmetic expression.

    Args:
        text: The expression string.

    Returns:
        Integer result of evaluation.
    """
    tokenizer = Tokenizer(text)
    tokens = tokenizer.tokenize()
    parser = Parser(tokens)
    ast = parser.parse()
    evaluator = Evaluator()
    return evaluator.evaluate(ast)
