def merge_intervals(intervals):
    """Merge a list of overlapping integer intervals.

    Each interval is a sequence of two integers (start, end) with
    start <= end. Returns a new list of merged, non-overlapping
    intervals sorted by start value.
    """
    if not intervals:
        return []

    sorted_intervals = sorted(intervals, key=lambda pair: pair[0])
    merged = [list(sorted_intervals[0])]

    for start, end in sorted_intervals[1:]:
        last = merged[-1]
        if start <= last[1]:
            last[1] = max(last[1], end)
        else:
            merged.append([start, end])

    return [tuple(interval) for interval in merged]
