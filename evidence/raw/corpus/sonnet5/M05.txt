"""A small least-recently-used (LRU) cache implementation.

This module provides a single class, LRUCache, backed by a dictionary
for O(1) lookups and a doubly linked list for O(1) reordering and
eviction of the least recently used entry.
"""


class _Node:
    """A single doubly linked list node holding a key/value pair."""

    __slots__ = ("key", "value", "prev", "next")

    def __init__(self, key=None, value=None):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None


class LRUCache:
    """A fixed-capacity cache that evicts the least recently used item.

    Both get and put run in O(1) time. Recency is tracked with a
    doubly linked list: the most recently used item sits right after
    the head sentinel, and the least recently used item sits right
    before the tail sentinel.
    """

    def __init__(self, capacity):
        if capacity <= 0:
            raise ValueError("capacity must be a positive integer")

        self.capacity = capacity
        self._map = {}

        # Sentinel nodes remove the need for special-casing empty lists.
        self._head = _Node()
        self._tail = _Node()
        self._head.next = self._tail
        self._tail.prev = self._head

    def get(self, key):
        """Return the value for key, or None if key is not present.

        A successful lookup marks the entry as most recently used.
        """
        node = self._map.get(key)
        if node is None:
            return None

        self._move_to_front(node)
        return node.value

    def put(self, key, value):
        """Insert or update key with value, evicting if over capacity."""
        existing = self._map.get(key)

        if existing is not None:
            existing.value = value
            self._move_to_front(existing)
            return

        if len(self._map) >= self.capacity:
            self._evict_least_recently_used()

        node = _Node(key, value)
        self._map[key] = node
        self._insert_at_front(node)

    def _insert_at_front(self, node):
        # Splice node in right after the head sentinel.
        first = self._head.next
        node.prev = self._head
        node.next = first
        self._head.next = node
        first.prev = node

    def _remove(self, node):
        # Unlink node from wherever it currently sits.
        node.prev.next = node.next
        node.next.prev = node.prev

    def _move_to_front(self, node):
        self._remove(node)
        self._insert_at_front(node)

    def _evict_least_recently_used(self):
        # The entry just before the tail sentinel is the LRU item.
        lru = self._tail.prev
        self._remove(lru)
        del self._map[lru.key]

    def __len__(self):
        return len(self._map)

    def __contains__(self, key):
        return key in self._map
