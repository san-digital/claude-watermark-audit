def merge_intervals(intervals):
    """Merge a list of overlapping integer intervals.

    Each interval is a two-element sequence [start, end] with start <= end.
    Intervals that overlap or merely touch (for example [1, 3] and [3, 5])
    are combined. The input is not modified.

    Returns a new list of disjoint [start, end] pairs sorted by start.
    """
    merged = []
    for start, end in sorted(intervals):
        if merged and start <= merged[-1][1]:
            merged[-1][1] = max(merged[-1][1], end)
        else:
            merged.append([start, end])
    return merged
